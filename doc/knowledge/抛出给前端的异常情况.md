package org.noLazy.common.aspect;

import lombok.extern.slf4j.Slf4j;
import org.noLazy.common.constant.ErrorCode;
import org.noLazy.common.domain.Result;
import org.noLazy.common.exception.*;
import org.noLazy.common.utils.WebUtils;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.validation.ObjectError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import org.springframework.validation.BindException;

import java.util.Objects;
import java.util.stream.Collectors;

/**
 * @ClassName: CommonExceptionAdvice
 * @Description: TODO
 * @Author: noLazy
 * @Date: 2026/3/17 15:20
 * @Version: 1.0.0
 * @ProjectName: IcedAmericanoMall
 * @Package: org.noLazy.common.aspect
 */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {


    /**
     * 处理自定义异常
     */
    @ExceptionHandler(CommonException.class)
    public Object handleCommonException(CommonException e) {
        ErrorCode errorCode = e.getErrorCode();
        int bizCode = errorCode.getCode();
        String message = e.getDisplayMessage();
        HttpStatus httpStatus = errorCode.getHttpStatus();
        log.error("自定义异常: httpStatus={}, bizCode={}, msg={}", httpStatus, bizCode, message, e);
        return processResponse(e);
    }
    /**
     * 处理数据库异常
     */
    @ExceptionHandler(DBException.class)
    public Object handleDBException(DBException e) {
        log.error("数据库异常: {}", e.getMessage(), e);
        return processResponse(e);
    }

    /**
     * 处理业务异常
     */
    @ExceptionHandler(BizException.class)
    public Object handleBizException(BizException e) {
        log.error("业务异常：{}", e.getMessage(), e);
        return processResponse(e);
    }

    /**
     * 处理禁止访问异常
     */
    @ExceptionHandler(ForbiddenException.class)
    public Object handleForbiddenException(ForbiddenException e) {
        log.error("禁止访问：{}", e.getMessage());
        return processResponse(e);
    }

    /**
     * 处理未授权异常
     */
    @ExceptionHandler(UnauthorizedException.class)
    public Object handleUnauthorizedException(UnauthorizedException e) {
        log.error("未授权异常: {}", e.getMessage());
        return processResponse(e);
    }

    /**
     * 处理参数校验异常（@Valid 抛出的异常）
     */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Object handleMethodArgumentNotValidException(MethodArgumentNotValidException e) {
        String msg = e.getBindingResult().getFieldErrors().stream()
                .map(ObjectError::getDefaultMessage)
                .collect(Collectors.joining("|"));
        log.error("参数校验异常：{}", msg, e);
        return processResponse(new BadRequestException(msg));
    }

    /**
     * 处理参数绑定异常（如普通表单提交校验）
     */
    @ExceptionHandler(BindException.class)
    public Object handleBindException(BindException e) {
        String msg = e.getBindingResult().getFieldErrors().stream()
                .map(FieldError::getDefaultMessage)
                .collect(Collectors.joining("；"));
        log.error("参数绑定异常：{}", msg, e);
        return processResponse(new BadRequestException(msg));
    }

    /**
     * 处理其他所有未捕获的异常（兜底）
     */
    @ExceptionHandler(Exception.class)
    public Object handleException(Exception e) {
        log.error("系统异常, url: {}", Objects.requireNonNull(WebUtils.getRequest()).getRequestURI(), e);
        return processResponse(new CommonException(ErrorCode.INTERNAL_ERROR, e.getMessage()));
    }

    /**
     * 将业务异常中的状态码映射为 HTTP 状态码
     * 如果你还想让代码更简洁，也可以只返回 R，但在网关层统一将非 200 的业务码转换为对应的 HTTP 状态码。不过直接在异常处理器中处理更直观。
     */
    private ResponseEntity<Result<Void>> processResponse(CommonException e) {
        return ResponseEntity.status(e.getErrorCode().getHttpStatus()).body(Result.error(e));
    }
}

在抛出异常时，**是否将详细异常信息返回给前端，取决于异常类型和场景**。基本原则是：

- **业务异常**（如验证码错误、手机号已注册）：可以返回**友好且具体**的提示信息，帮助用户理解问题。这些信息通常来自枚举的默认消息，或者业务代码中提供的自定义消息（例如“手机号 138****1234 已被注册”）。
- **系统异常**（如数据库连接失败、空指针）：**绝不能**将内部错误细节（如 SQL 异常、堆栈信息）直接返回给前端，应统一返回“服务器内部错误，请稍后再试”等通用提示。

你当前的代码中，`CommonException` 已经支持通过构造方法传入 `customMessage`，并且 `getDisplayMessage()` 会优先返回自定义消息，这是很好的设计。但在全局异常处理的兜底分支（`handleException`）中，存在一个**安全隐患**：

```java
@ExceptionHandler(Exception.class)
public Object handleException(Exception e) {
    log.error("系统异常, url: {}", ...);
    return processResponse(new CommonException(ErrorCode.INTERNAL_ERROR, e.getMessage()));
}
```

这里把原始异常 `e.getMessage()` 直接作为自定义消息传给了 `CommonException`，最终会返回给前端。如果发生 `NullPointerException` 或 `SQLException`，其 `getMessage()` 可能包含敏感信息（如表名、字段名、代码位置），这会造成信息泄露。

---

## 推荐做法

### 1. 业务异常（BizException / CommonException）
- **默认使用枚举中预定义的友好消息**（如 `CAPTCHA_ERROR` → “验证码错误”）。
- **必要时使用自定义消息**，但需确保不泄露敏感数据（例如注册时提示“手机号 138****0000 已被注册”可以对手机号脱敏）。
- 前端只需要根据业务码做差异化处理，消息仅用于展示。

### 2. 系统异常（兜底 Exception）
- **只返回通用错误提示**，如 `ErrorCode.INTERNAL_ERROR` 的默认消息（“服务器内部错误”）。
- **不要**把原始异常的消息传递出去。
- 详细错误信息只记录在**服务端日志**中，供运维排查。

---

## 修改建议

修改 `GlobalExceptionHandler` 中的 `handleException` 方法：

```java
@ExceptionHandler(Exception.class)
public Object handleException(Exception e) {
    log.error("系统异常, url: {}", Objects.requireNonNull(WebUtils.getRequest()).getRequestURI(), e);
    // 只返回通用错误码和消息，不暴露原始异常信息
    return processResponse(new CommonException(ErrorCode.INTERNAL_ERROR));
}
```

同时，确保 `ErrorCode.INTERNAL_ERROR` 的消息为：“服务器内部错误，请稍后再试”（或类似友好文案）。

---

## 总结

| 异常类型 | 返回给前端的信息 | 示例 |
|---------|----------------|------|
| 业务异常（验证码错误、手机号已注册等） | 枚举默认消息或自定义友好消息 | “验证码错误”、“手机号 138****0000 已被注册” |
| 系统异常（空指针、DB 异常等） | 统一通用消息 | “服务器内部错误，请稍后再试” |

**核心原则**：前端只接收**用户能理解且不泄露系统内部细节**的信息；所有技术细节、堆栈信息都只记录在服务端日志中。这样既保证了用户体验，又维护了系统安全。
